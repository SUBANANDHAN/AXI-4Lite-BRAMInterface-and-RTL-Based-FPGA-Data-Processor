module tb_axi_bram();
// System Signals
reg s_axi_aclk_0 = 0;
reg s_axi_aresetn_0 = 0;

// AXI Write Address Channel
reg [31:0] s_axi_awaddr = 0;
reg [2:0] s_axi_awprot = 0;
reg s_axi_awvalid = 0;
wire s_axi_awready;

// AXI Write Data Channel
reg [31:0] s_axi_wdata = 0;
reg [3:0] s_axi_wstrb = 4'hF;
reg s_axi_wvalid = 0;
wire s_axi_wready;

// AXI Write Response Channel
wire [1:0] s_axi_bresp;
wire s_axi_bvalid;
reg s_axi_bready = 0;

// AXI Read Address Channel
reg [31:0] s_axi_araddr = 0;
reg [2:0] s_axi_arprot = 0;
reg s_axi_arvalid = 0;
wire s_axi_arready;

// AXI Read Data Channel
wire [31:0] s_axi_rdata;
wire [1:0] s_axi_rresp;
wire s_axi_rvalid;
reg s_axi_rready = 0;

// Clock Generation (100 MHz)
always #5 s_axi_aclk_0 = ~s_axi_aclk_0;

// Instantiate the Wrapper
// Ensure "design_1_wrapper" matches your top-level file name
design_1_wrapper dut (
    .s_axi_aclk_0(s_axi_aclk_0),
    .s_axi_aresetn_0(s_axi_aresetn_0),
    .S_AXI_0_awaddr(s_axi_awaddr),
    .S_AXI_0_awprot(s_axi_awprot),
    .S_AXI_0_awvalid(s_axi_awvalid),
    .S_AXI_0_awready(s_axi_awready),
    .S_AXI_0_wdata(s_axi_wdata),
    .S_AXI_0_wstrb(s_axi_wstrb),
    .S_AXI_0_wvalid(s_axi_wvalid),
    .S_AXI_0_wready(s_axi_wready),
    .S_AXI_0_bresp(s_axi_bresp),
    .S_AXI_0_bvalid(s_axi_bvalid),
    .S_AXI_0_bready(s_axi_bready),
    .S_AXI_0_araddr(s_axi_araddr),
    .S_AXI_0_arprot(s_axi_arprot),
    .S_AXI_0_arvalid(s_axi_arvalid),
    .S_AXI_0_arready(s_axi_arready),
    .S_AXI_0_rdata(s_axi_rdata),
    .S_AXI_0_rresp(s_axi_rresp),
    .S_AXI_0_rvalid(s_axi_rvalid),
    .S_AXI_0_rready(s_axi_rready)
);

initial begin
    // Reset sequence
    s_axi_aresetn_0 = 0;
    #100;
    s_axi_aresetn_0 = 1;
    #50;

    //--- (a) Write Transaction
    @(posedge s_axi_aclk_0);
    s_axi_awaddr <= 32'h0000_0000; // Single address
    s_axi_awvalid <= 1;
    s_axi_wdata <= 32'hAAAA_BBBB; // Data to write
    s_axi_wvalid <= 1;
    s_axi_bready <= 1;
    wait(s_axi_awready && s_axi_wready);
    @(posedge s_axi_aclk_0);
    s_axi_awvalid <= 0;
    s_axi_wvalid <= 0;
    wait(s_axi_bvalid);
    @(posedge s_axi_aclk_0);
    s_axi_bready <= 0;
    $display("Write Operation Complete.");
    #100;

    // --- (b) Read Back Transaction ---
    @(posedge s_axi_aclk_0);
    s_axi_araddr <= 32'h0000_0000; // Same address
    s_axi_arvalid <= 1;
    s_axi_rready <= 1;
    wait(s_axi_arready);
    @(posedge s_axi_aclk_0);
    s_axi_arvalid <= 0;
    wait(s_axi_rvalid);
    $display("Read Back Data: %h", s_axi_rdata);
    @(posedge s_axi_aclk_0);
    s_axi_rready <= 0;
    #200;
    $finish;
end
endmodule
