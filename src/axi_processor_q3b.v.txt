module axi_processor_q3b (
    input wire clk,
    input wire resetn,
    // AXI Read
    output reg [14:0] m_axi_araddr,
    output reg m_axi_arvalid,
    input wire m_axi_arready,
    input wire [31:0] m_axi_rdata,
    input wire m_axi_rvalid,
    output reg m_axi_rready,
    // AXI Write
    output reg [14:0] m_axi_awaddr,
    output reg m_axi_awvalid,
    input wire m_axi_awready,
    output reg [31:0] m_axi_wdata,
    output reg [4:0] m_axi_wstrb,
    output reg m_axi_wvalid,
    input wire m_axi_wready,
    input wire m_axi_bvalid,
    output reg m_axi_bready,
    // Debug
    output wire [15:0] debug_a,
    output wire [15:0] debug_b,
    output wire [31:0] debug_mult_out,
    output wire [2:0] debug_state
);

localparam IDLE = 0, READ_ADDR = 1, READ_DATA = 2, CALC = 3, WRITE_ADDR = 4, WRITE_RESP = 5, INC_ADDR = 6, DONE = 7;
reg [2:0] state;
reg [31:0] captured_data;
reg [31:0] rd_addr;
reg [31:0] wr_addr;
wire [31:0] multiplier_out;

dadda_16x16_signed_method2 mul (
    .a(captured_data[31:16]),
    .b(captured_data[15:0]),
    .y(multiplier_out)
);

assign debug_a = captured_data[31:16];
assign debug_b = captured_data[15:0];
assign debug_mult_out = multiplier_out;
assign debug_state = state;

localparam START_ADDR = 32'h0000_0000;
localparam LAST_INPUT_ADDR = 32'h0000_003C;
localparam WRITE_BASE = 32'h0000_0040;

always @(posedge clk) begin
    if (!resetn) begin
        state <= IDLE;
        rd_addr <= START_ADDR;
        wr_addr <= WRITE_BASE;
        m_axi_araddr <= 0;
        m_axi_arvalid <= 0;
        m_axi_rready <= 0;
        m_axi_awaddr <= 0;
        m_axi_awvalid <= 0;
        m_axi_wdata <= 0;
        m_axi_wvalid <= 0;
        m_axi_wstrb <= 4'b1111;
        m_axi_bready <= 0;
        captured_data <= 0;
    end else begin
        case (state)
            IDLE: begin
                state <= READ_ADDR;
            end
            READ_ADDR: begin
                m_axi_araddr <= rd_addr;
                m_axi_arvalid <= 1;
                if (m_axi_arvalid && m_axi_arready) begin
                    m_axi_arvalid <= 0;
                    m_axi_rready <= 1;
                    state <= READ_DATA;
                end
            end
            READ_DATA: begin
                if (m_axi_rvalid && m_axi_rready) begin
                    captured_data <= m_axi_rdata;
                    m_axi_rready <= 0;
                    state <= CALC;
                end
            end
            CALC: begin
                m_axi_awaddr <= wr_addr;
                m_axi_wdata <= multiplier_out; 
                m_axi_awvalid <= 1;
                m_axi_wvalid <= 1;
                state <= WRITE_ADDR;
            end
            WRITE_ADDR: begin
                if (m_axi_awvalid && m_axi_awready) m_axi_awvalid <= 0;
                if (m_axi_wvalid && m_axi_wready) m_axi_wvalid <= 0;
                if ((!m_axi_awvalid || m_axi_awready) && (!m_axi_wvalid || m_axi_wready)) begin
                    m_axi_bready <= 1;
                    state <= WRITE_RESP;
                end
            end
            WRITE_RESP: begin
                if (m_axi_bvalid && m_axi_bready) begin
                    m_axi_bready <= 0;
                    state <= INC_ADDR;
                end
            end
            INC_ADDR: begin
                if (rd_addr >= LAST_INPUT_ADDR) begin
                    state <= START_ADDR;
                end else begin
                    rd_addr <= rd_addr + 4;
                    wr_addr <= wr_addr + 4;
                    state <= IDLE;
                end
            end
            DONE: begin
                state <= DONE;
            end
            default: state <= IDLE;
        endcase
    end
end
endmodule
